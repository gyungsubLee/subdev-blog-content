### 구현1) 이중 배열(int&#91;&#93;&#91;&#93;)

<a id="1c3eda9b0fe181a19e74fd840a435413"></a>

```java
import java.lang.reflect.Array;
import java.util.*;

public class Main {
    static int n, m;
    static int[][] graph;


    public static void main(String args[]) {
        Main T = new Main();
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        graph = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            graph[a][b] = 1;
            graph[b][a] = 1;
        }
        System.out.println(graph);
        System.out.println();
    }
}
```

### 구현2) 이중 ArrayList

<a id="1c3eda9b0fe1815ea78ef99d532f365e"></a>

```java
import java.lang.reflect.Array;
import java.util.*;

public class Main {
    static int n, m;
    static ArrayList<ArrayList<Integer>> graph;


    public static void main(String args[]) {
        Main T = new Main();
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        graph = new ArrayList<ArrayList<Integer>>();
        for(int i=0; i<=n; i++){
            graph.add(new ArrayList<Integer>());
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            graph.get(a).add(b);
            graph.get(b).add(a);
        }
        System.out.println(graph);
        System.out.println();
    }
}
```

import java.lang.reflect.Array;

public class Main {
    static int n, m;
    static int[][] graph;


    public static void main(String args[]) {
        Main T = new Main();
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        graph = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
