```javascript
// subText - 애니메이션
gsap.utils.toArray('.subText p').forEach((selector) => {
    console.log(selector)
    gsap.timeline({
        scrollTrigger: {
            trigger: selector,
            start: '0% 100%',
            end: '100% 100%',
            scrub: 1,
            markers: true
        }
    })
    .fromTo(
        selector,
        {overflow: 'hidden', x: 20, opacity: 0},
        {x:0, opacity: 1, ease: 'none', duration: 5},
        0)
})
```
