```bash
# 기본 모드 전환
i   # 입력 모드로 전환  
ESC # 명령 모드로 전환  

# 커서 이동
h   # 왼쪽 이동  
l   # 오른쪽 이동  
j   # 아래 이동  
k   # 위 이동  

# 텍스트 편집
x   # 현재 커서의 문자 삭제  
dd  # 현재 줄 삭제  
yy  # 현재 줄 복사  
p   # 붙여넣기  

# 저장 및 종료
:w   # 저장  
:q   # 종료  
:wq  # 저장 후 종료  
:q!  # 저장 없이 강제 종료  

# 검색
/텍스트  # 텍스트 검색  
n        # 다음 검색 결과로 이동  
N        # 이전 검색 결과로 이동  
```
