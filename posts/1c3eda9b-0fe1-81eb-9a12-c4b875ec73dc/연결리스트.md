class LinkedList {
    private static Node head = null;

    // 생성
    public void add(int value) {
        // LinkedList 초기화
        if(head == null){
            head = new Node(value, null);
            return;
        }

        Node node = head;
        while(node.next != null){
            node = node.next;
        }

        node.next = new Node(value, null);
    }

    // 조회
    public int get(int index){
        if(head == null){
            System.out.println("조회할 노드개 존재하지 않습니다.");
            return -1;
        }

        Node node = head;
        while(index>0){
            node = node.next;
            index--;
        }


        return node.val;
    }


    // 수정
    public void update(int index, int value){
        if(index < 0) {
            System.out.println("index는 0이상부터 가능합니다.");
            return;
        }

        int prtIdx = index;

        Node node = head;

        while(index>0){
            if(node.next==null){
                System.out.println("해당하는 index = " + prtIdx + " 의 노드는 존재하지 않습니다.");
                return;
            }
            node = node.next;
            index--;
        }
        node.val=value;

        System.out.println(prtIdx + "번째 node 값이 " + value +"로 수정 됐습니다.");
    }

    // 제거
    public int remove(){
        if(head==null){
            System.out.println("삭제할 노드가 존재하지 않습니다.");
            return -1;
        }
        Node node = head;

        head = node.next;
        return node.val;
    }
}
