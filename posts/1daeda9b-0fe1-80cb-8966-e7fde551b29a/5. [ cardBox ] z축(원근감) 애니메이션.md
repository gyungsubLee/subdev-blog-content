### GSAP 인터렉티브 애니메이션 효과

<a id="1daeda9b0fe1801d954cdfd589aa6c54"></a>

```javascript
// con4 CardBox - z 축 애니메이션
    gsap.utils.toArray('.con4 .listBox .cardBox').forEach((selector) => {
        gsap.timeline({
            scrollTrigger: {
                trigger: selector,
                start: '0% 20%',
                end: '0% 0%',
                scrub: 1,
                markers: true
            }
        })
        .to( selector, { 
		        rotateX: -10, 
		        scale: 0.9, 
		        transformOrigin: 'top', 
		        filter: 'brightness(0.3)',   // 기본값으로 filter: brightness(1) 설정을 해야함
	        }, 0)
        // transformOrigin: 애니메이션 기준점
          // top을 주면 top 기준으로 애니메이션 진행
          // ex) roateX -> top을 기준으로 아래가 회전함
    })
```
