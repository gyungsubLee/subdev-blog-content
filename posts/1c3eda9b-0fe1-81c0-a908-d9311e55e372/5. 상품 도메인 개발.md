```java
    }

    public List<Item> findItems() {
```

<summary><strong>&#91; 코드 &#93; 저장소 &#45; ItemRepository</strong></summary>

- save 로직에서는 인자로 받는 Item 값에서 id 유무에 따라 영속 상태를 구분한다&#46; <br>id 값이 존재하는 경우, 이미 JPA에서 해당 값을 자동 생성하여 DB에 저장한 <u>영속 상태</u>이므로<br><code>merge()</code>를 통해 업데이트 처리만 하면 된다&#46; <br>

<a id="1c3eda9b0fe181338f09fd4be4c20f32"></a>

```java
package com.prac.jpa1.repository;

import com.prac.jpa1.domain.Item;
import jakarta.persistence.EntityManager;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Repository;

import java.util.List;

@Repository
@RequiredArgsConstructor
public class ItemRepository {

    private final EntityManager em;

    public void save(Item item) {
        if (item.getId() == null) {
            em.persist(item);
        } else {
            em.merge(item);
        }
    }

    public Item findOne(Long id) {
        return em.find(Item.class, id);
    }

    public List<Item> findAll(){
        return em.createQuery("select i from Item i", Item.class)
                .getResultList();
    }
}
```
