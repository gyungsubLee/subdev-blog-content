고루틴 간의 데이터를 교환해야할 경우가 빈번하다 

<a id="3caeda9b0fe18099b00de731203f679b"></a>

그러나 각자 실행되기 때문에 callback 과 같은 복잡한 문법을 사용해야 하는데 <br>이를 Go에서는 <code>채널</code> 이라는 개념을 만들어 해결한다&#46; 

<a id="3caeda9b0fe180578e6fdc6a9c879a66"></a>

고채널을 통해 고루틴 간의 데이터 교환이 용이하며 통신의 흐름을 명확하게 표현할 수 있어 코드의 복잡성을 낮춘다&#46; 

<a id="3caeda9b0fe18060861be0811de96e03"></a>

### ex) 크롤링 작업

![](7a1eb1ad600f5338.webp)

- 다수에 사이트에서 정보를 가져오는 크롤링 작업 시, 여러 사이트 수집을 각각 고루틴으로 병렬 처리 후, 고채널을 통해 <strong><u>안전하게 메인 루인으로 받</u></strong>아올 수 있다&#46; 

<a id="3caeda9b0fe1802291b7c964622cb6bb"></a>

### 사용법

1. 채널 생성
    <a id="3caeda9b0fe180a09278eb2074dd71c8"></a>

    ```go
    채널 이름 := make(chan  타입)
    ```

    - 채널 생성 이때 데이터 타입 명시 (선언된 타입 형식만 데이터 통과 가능)

1. 채널 송수신
    <a id="3caeda9b0fe1800cb484ececf2414ddf"></a>

    ```go
    // 데이터 송신
    채널_이름 <- 데이터 

    // 데이터 수신
    변수 := <- 채널_이름 

    // 수신할 변수 없이도 가능
    <- 채널_이름 
    ```

1. 채널 닫기
    <a id="3caeda9b0fe180beb5dbe7c8ecfc68ee"></a>

    ```go
    close(채널_이름) // 해당 채널을 더이상 사용하지 않음
    ```

    - 채널을 닫으면 해당 채널로 데이터 전송이 불가능함

<summary>ex1) 채널 기본 사용</summary>

<a id="3caeda9b0fe180908ae0d9528a20915d"></a>

```go
package main

import (
	"fmt"
	"time"
)

func task(id string, ch chan string) {
	fmt.Printf("작업 %s : 시작\n", id)
	time.Sleep(time.Second)  // 작업 1초 수행
	fmt.Printf("작업 %s : 꿑\n", id)

	// 채널에 데이터 전송
	// 채널 <- 데이터 (채널에서 선언한 타입과 실제 전송하는 데이터 타입이 일치해야 함)
	ch <- id
}


func main() {
	// 채널 
	// 고루틴 간의 통신을 위한 도구로, 데이터를 안전하게 주고받을 수 있음
	
	ch := make(chan string)

	go task("1", ch)
	go task("2", ch)
	go task("3", ch)
	
	id := <- ch // 채널에서 데이터 수신
	fmt.Printf("task %s의 작업 종료 데어터를 받음\n", id)

	id = <- ch // 채널에서 데이터 수신
	fmt.Printf("task %s의 작업 종료 데어터를 받음\n", id)

	id = <- ch // 채널에서 데이터 수신
	fmt.Printf("task %s의 작업 종료 데어터를 받음\n", id)
	
	fmt.Printf("모든 작업 종료")

}
```

<summary>ex3) 효과적인 사용&#58; <code>for&#45;range</code>, <code>ch&#46;close</code></summary>

```go
package main

import "fmt"

func getNumbers(ch chan int) {
	for i := 1; i <= 5; i++ {
		ch <- i  // 채널에 숫자 전송
	}

	close(ch)
}

func main() {
	ch := make(chan int, 5)

	go getNumbers(ch) // 고루틴에서 채널로 숫자 1 ~ 5 전송

	/* range 
		range를 사용하여 채널에서 값을 받음
		- 채널을 닫을 때까지 반복해서 값을 꺼냄
		- 채널을 닫지 않으면 데드락 발생
	*/
	for num := range ch {
		fmt.Println("받은 값: ", num)
	}

	fmt.Println("채널 전송 테스트 종료")

}
```

<summary>ex4) <code>select</code>, <code>ok</code></summary>

<a id="3caeda9b0fe1802ca649c8cdf7ab27b2"></a>

```go
package main

import (
	"fmt"
	"time"
)

func getNumbers(ch chan<- int, id int, count int) {
	for i := 1; i <= count; i++ {
		fmt.Printf("고루틴 %d에서 %d 보냄 - ", id, i)
		ch <- i // 채널 숫자 전송
		time.Sleep(time.Microsecond * 500)
	}

	close(ch)
}

func main() {
    ch1 := make(chan int)
    ch2 := make(chan int)

    go getNumbers(ch1, 1, 5)
    go getNumbers(ch2, 2, 3)

    for ch1 != nil || ch2 != nil {
        select {
        case num, ok := <-ch1:
		        
		        /* 
		        select 구문에서 닫힌 채널 제거 
						채널이 닫히면 num = 0, ok = false 가 됨
						*/
            if !ok {
                ch1 = nil
                continue
            }
            fmt.Println("채널 1에서 받은 값:", num)

        case num, ok := <-ch2:
            if !ok {
                ch2 = nil
                continue
            }
            fmt.Println("채널 2에서 받은 값:", num)
        }
    }

    fmt.Println("모든 고루틴 완료")
}
```
